# ===== NOTE PER SVILUPPATORI =====
# 
# Parametri supportati per DragDrop + AssemblyMode:
#
# - AssemblyMode=true                    : Abilita modalità assemblaggio
# - AssemblyConfig=percorso_file.json    : Percorso configurazione assemblaggio
# - CurrentStep=nome_step                : Step corrente nell'assemblaggio
# - AllowedComponents=comp1,comp2        : Lista componenti montabili nello step
# - RequiredPrevious=step1,step2         : Step prerequisiti per questo step
# - InterchangeableNode=node_name        : Nome nodo intercambiabile
# - ShowSnapPoints=comp1,comp2           : Componenti per cui mostrare snap points
# - MinimumRequired=numero              : Numero minimo componenti da montare
# - MaximumAllowed=numero               : Numero massimo componenti montabili
# - UndoEnabled=true/false              : Abilita undo per questo step
# - ValidateAssembly=true               : Valida assemblaggio alla fine dello step
# - ParallelAssembly=true               : Permette montaggio parallelo componenti
# - ShowAssemblyStatus=true             : Mostra status assemblaggio nell'UI
# - AutoAdvance=true/false              : Avanzamento automatico al completamento
#
# Integrazione con sistema esistente:
# - Tutti i parametri esistenti (Elemento, Descrizione, Azione1, etc.) continuano a funzionare
# - I parametri Assembly sono additivi e non sostituiscono funzionalità esistenti
# - Compatibilità completa con tutorial non-assembly esistenti
