# Tutorial di Esempio - Sistema Drag & Drop 3D
# Questo file dimostra come configurare tutorial con funzionalità drag & drop

[Tutorial Drag & Drop - Assemblaggio Componenti]
Descrizione=Tutorial per imparare il sistema drag & drop con assemblaggio componenti
CameraPos=(5,3,5)
CameraTarget=(0,0,0)
CameraZoom=6.0

[Step 1 - Abilitazione Drag & Drop]
Descrizione=Abilita il sistema drag & drop per tutti gli oggetti
Elemento=filtro.glb
DragDrop=true
DragDropDistance=1.2
# Note: DragDrop=true abilita il sistema, DragDropDistance imposta la distanza di snap

[Step 2 - Drag & Drop Selettivo]
Descrizione=Abilita il drag & drop solo per oggetti specifici
Elemento=tappino_grasso_sx
DragDrop=true
DragDropObjects=filtro,tappino_grasso_sx,vite_coperchio
DragDropDistance=0.8
# Note: DragDropObjects specifica una lista di oggetti draggabili

[Step 3 - Interazione Normale]
Descrizione=Disabilita il drag & drop e usa solo click normali
Elemento=ingrassatore
DragDrop=false
# Note: DragDrop=false disabilita completamente il sistema

[Step 4 - Snap Preciso]
Descrizione=Riabilita drag & drop con snap molto preciso
Elemento=filtro.glb
DragDrop=true
DragDropDistance=0.5
# Note: Distanza snap ridotta per maggiore precisione

[Step 5 - Assemblaggio Finale]
Descrizione=Posiziona tutti i componenti nelle loro posizioni finali
DragDrop=true
DragDropObjects=filtro,tappino_grasso_sx,tappino_grasso_dx,ingrassatore
DragDropDistance=1.0
CameraPos=(3,2,3)
CameraTarget=(0,0,0)

[Tutorial Avanzato - Meccanismi Complessi]
Descrizione=Tutorial avanzato per meccanismi con molte parti
CameraPos=(4,4,6)
CameraZoom=8.0

[Step 1 - Setup Iniziale]
Descrizione=Prepara l'ambiente per assemblaggio complesso
DragDrop=true
DragDropDistance=1.5
# Distanza snap più ampia per parti grandi

[Step 2 - Assemblaggio Preciso]
Descrizione=Assemblaggio componenti di precisione
DragDrop=true
DragDropObjects=vite_piccola,rondella,dado
DragDropDistance=0.3
# Snap molto preciso per componenti piccoli

[Step 3 - Controllo Qualità]
Descrizione=Verifica posizionamento senza modifiche
DragDrop=false
# Disabilita drag & drop per fase di controllo

# ===== RIFERIMENTO PARAMETRI DRAG & DROP =====

# DragDrop=true/false
#   - true: abilita il sistema drag & drop per questo step
#   - false: disabilita il sistema drag & drop per questo step

# DragDropObjects=oggetto1,oggetto2,oggetto3
#   - Lista di oggetti draggabili separati da virgole
#   - Se non specificato, tutti gli oggetti diventano draggabili
#   - I nomi oggetti sono senza estensioni (.glb, .obj, etc.)

# DragDropDistance=valore
#   - Distanza in unità 3D per attivazione snap automatico
#   - Valori tipici: 0.3-2.0
#   - 0.3: snap molto preciso (parti piccole)
#   - 1.0: snap normale (default)
#   - 2.0: snap ampio (parti grandi)

# ===== ESEMPI DI USO PRATICO =====

# Per assemblaggio meccanico preciso:
# DragDrop=true
# DragDropObjects=vite,dado,rondella
# DragDropDistance=0.4

# Per posizionamento parti grandi:
# DragDrop=true
# DragDropDistance=1.8

# Per disabilitare completamente:
# DragDrop=false

# ===== INTEGRAZIONE CON ALTRI SISTEMI =====

# Il drag & drop si integra automaticamente con:
# - Sistema di animazioni esistente
# - Highlighting degli oggetti
# - Controlli camera
# - Sistema tutorial steps

# Durante il drag, il sistema click normale è disabilitato automaticamente
# Al termine del drag, il sistema click viene riabilitato

# ===== NOTE TECNICHE =====

# 1. Il sistema memorizza automaticamente le posizioni originali di tutti gli oggetti
# 2. Lo snap avviene verso il centro del bounding box originale dell'oggetto
# 3. Il feedback visivo (cerchi verdi) appare solo durante il drag
# 4. Gli oggetti snappano con animazione fluida quando rilasciati nella zona
# 5. Il sistema è completamente disabilitabile per compatibilità con tutorial esistenti
