# Tutorial per Manutenzione Pompa del Vuoto - MODIFICATO con Drag & Drop
# Step 1 modificato per usare drag & drop invece di animazioni automatiche

[Allineamento]
  CameraPos=(-1.5, 0.5,-0.2)
  CameraTarget=(0, 0, 0)
  CameraZoom=1.2
  CameraTransitionTime=2.5

  # Posiziona l'ingrassatore in posizione iniziale scomposta
  Posizione=ingrassatore:(1.5,0.2,-0.5)   # Posizione iniziale (la tua attuale)
  Rotazione=ingrassatore:(0,90,0)          # Rotazione iniziale (la tua attuale)

  [Step 1 - Riposiziona l'ingrassatore con Drag & Drop]
  Elemento=models/ingrassatore.glb
  Utensile=Mani
  Descrizione=Trascina l'ingrassatore nella sua posizione originale. Rilascialo quando è vicino al centro della scena per il posizionamento automatico.
  
  # Abilita drag & drop per questo step
  DragDrop=true
  DragDropObjects=ingrassatore
  DragDropDistance=1.2                     # Distanza di snap generosa
  
  # Posizione e rotazione TARGET (coordinate finali corrette)
  # SOSTITUISCI QUESTE con le coordinate che vuoi ottenere:
  TargetPosizione=ingrassatore:(0.0,0.0,0.0)    # Centro scena
  TargetRotazione=ingrassatore:(0,0,0)           # Rotazione neutra
  
  # RIMUOVI le azioni automatiche originali:
  # Azione1=centro:(-0.2,0,0);rotazione:(0,0,90,0.8)
  # Azione2=traslazione:(-0.2,0.0,0,0.8)
  # Azione3=appoggia(1.5)

[Step 2- Prima vite]
  Elemento=models/vite_coperchio_1.glb
  Utensile=Mani
  Descrizione=svita la manopola
  Azione1=svita
  
  # Disabilita drag & drop per i prossimi step
  DragDrop=false
